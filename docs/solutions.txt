'use client'

import { useMemo } from 'react'
import { Bell, Loader2, Brain, AlertCircle, Gauge } from 'lucide-react'
import Link from 'next/link'

import { useAthletes } from '@/context/AthleteContext'
import { useAuth } from '@/context/AuthContext'

// DESCOMENTE DEPOIS
// import { useAlerts } from '@/context/AlertContext'
// import { KpiCard } from '@/components/kicker/kpi-card'
// import { Avatar } from '@/components/kicker/avatar'
// import { formatNumber, formatDelta } from '@/lib/utils'

import {
  ResponsiveContainer,
  RadarChart,
  Radar,
  PolarGrid,
  PolarAngleAxis,
  ScatterChart,
  Scatter,
  XAxis,
  YAxis,
  ZAxis,
  Tooltip,
  Cell,
  AreaChart,
  Area,
  CartesianGrid,
} from 'recharts'

const CustomTooltip = ({ active, payload, label }: any) => {
  if (!active || !payload?.length) return null

  return (
    <div className="bg-zinc-900 border border-zinc-800 rounded-xl p-3 text-white">
      <p className="text-xs font-bold mb-1">{label}</p>

      {payload.map((entry: any, i: number) => (
        <div key={i} className="text-xs">
          {entry.name}: {entry.value}
        </div>
      ))}
    </div>
  )
}

export default function DashboardPage() {
  const { loading: authLoading } = useAuth()
  const { athletes, loading: athletesLoading, error } = useAthletes()

  // DESCOMENTE DEPOIS
  // const { activeCount } = useAlerts()

  const activeCount = 0

  const stats = useMemo(() => {
    if (!athletes.length) {
      return {
        avgSpeed: 0,
        avgSprintDist: 0,
        avgLoad: 0,
        avgPse: 0,
        speedDelta: 0,
        sprintDelta: 0,
        loadDelta: 0,
        pseDelta: 0,
      }
    }

    return {
      avgSpeed:
        athletes.reduce((acc, a) => acc + a.speed, 0) /
        athletes.length,

      avgSprintDist: Math.round(
        athletes.reduce(
          (acc, a) => acc + a.sprintDistance,
          0
        ) / athletes.length
      ),

      avgLoad: Math.round(
        athletes.reduce(
          (acc, a) => acc + a.weeklyLoad,
          0
        ) / athletes.length
      ),

      avgPse:
        athletes.reduce((acc, a) => acc + a.pse, 0) /
        athletes.length,

      speedDelta:
        athletes.reduce(
          (acc, a) => acc + a.speedDelta,
          0
        ) / athletes.length,

      sprintDelta:
        athletes.reduce(
          (acc, a) => acc + a.sprintDelta,
          0
        ) / athletes.length,

      loadDelta:
        athletes.reduce(
          (acc, a) => acc + a.loadDelta,
          0
        ) / athletes.length,

      pseDelta:
        athletes.reduce(
          (acc, a) => acc + a.pseDelta,
          0
        ) / athletes.length,
    }
  }, [athletes])

  const topPerformers = useMemo(() => {
    return [...athletes]
      .sort((a, b) => b.speed - a.speed)
      .slice(0, 3)
  }, [athletes])

  const squadRadarAvg = useMemo(() => {
    if (!athletes.length) return []

    const total = athletes.reduce(
      (acc, athlete) => ({
        velocidade:
          acc.velocidade +
          athlete.radar.velocidade,

        resistencia:
          acc.resistencia +
          athlete.radar.resistencia,

        explosividade:
          acc.explosividade +
          athlete.radar.explosividade,

        carga:
          acc.carga +
          athlete.radar.carga,

        recuperacao:
          acc.recuperacao +
          athlete.radar.recuperacao,

        tecnica:
          acc.tecnica +
          athlete.radar.tecnica,
      }),
      {
        velocidade: 0,
        resistencia: 0,
        explosividade: 0,
        carga: 0,
        recuperacao: 0,
        tecnica: 0,
      }
    )

    const count = athletes.length

    return [
      {
        subject: 'Velocidade',
        A: total.velocidade / count,
      },
      {
        subject: 'Resistência',
        A: total.resistencia / count,
      },
      {
        subject: 'Explosão',
        A: total.explosividade / count,
      },
      {
        subject: 'Carga',
        A: total.carga / count,
      },
      {
        subject: 'Recuperação',
        A: total.recuperacao / count,
      },
      {
        subject: 'Técnica',
        A: total.tecnica / count,
      },
    ]
  }, [athletes])

  const riskData = useMemo(() => {
    return athletes.map(a => ({
      name: a.name,
      load: a.weeklyLoad,
      pse: a.pse,
      z: 10,
    }))
  }, [athletes])

  const loadEvolution = useMemo(() => {
    return [...athletes]
      .sort((a, b) => a.weeklyLoad - b.weeklyLoad)
      .map((athlete, index) => ({
        jornada: `A${index + 1}`,
        carga: athlete.weeklyLoad,
      }))
  }, [athletes])

  const isLoading =
    authLoading ||
    (athletesLoading && athletes.length === 0)

  if (isLoading) {
    return (
      <div className="flex items-center justify-center h-full">
        <Loader2
          size={24}
          className="animate-spin"
        />
      </div>
    )
  }

  return (
    <div className="flex flex-col bg-[var(--surface-1)] min-h-screen">

      {/* HEADER ANTIGO */}
      <div className="h-12 flex items-center justify-between px-4 border-b border-[var(--border-subtle)] sticky top-0 bg-[var(--surface-1)] z-10">
        <div>
          <div className="text-sm font-medium">
            Kicker IQ Dashboard
          </div>

          <div className="text-[10px] opacity-60">
            {new Date().toLocaleDateString()}
          </div>
        </div>

        <Link
          href="/alertas"
          className="relative w-8 h-8 flex items-center justify-center rounded-lg border"
        >
          <Bell size={14} />

          {activeCount > 0 && (
            <span className="absolute top-0 right-0 w-2 h-2 bg-red-500 rounded-full" />
          )}
        </Link>
      </div>

      <div className="p-4 flex flex-col gap-6">

        {error && (
          <div className="bg-red-500/10 text-red-500 p-3 rounded-xl">
            {error}
          </div>
        )}

        {/* KPIS */}

        {/* DESCOMENTE QUANDO REATIVAR KpiCard */}

        {/*
        <section className="grid grid-cols-2 lg:grid-cols-4 gap-4">

          <KpiCard
            label="Velocidade Média"
            value={`${formatNumber(stats.avgSpeed,1)} km/h`}
            delta={formatDelta(stats.speedDelta)}
            deltaDirection="up"
          />

          <KpiCard
            label="Sprint Médio"
            value={`${stats.avgSprintDist} m`}
            delta={formatDelta(stats.sprintDelta)}
            deltaDirection="up"
          />

          <KpiCard
            label="Carga Semanal"
            value={`${stats.avgLoad} AU`}
            delta={formatDelta(stats.loadDelta)}
            deltaDirection="up"
          />

          <KpiCard
            label="PSE Média"
            value={stats.avgPse.toFixed(1)}
            delta={formatDelta(stats.pseDelta)}
            deltaDirection="up"
          />

        </section>
        */}

        {/* KPIS PROVISÓRIO */}

        <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">

          <div className="p-4 rounded-2xl border">
            <p className="text-xs opacity-60">
              Velocidade Média
            </p>
            <p className="font-bold">
              {stats.avgSpeed.toFixed(1)} km/h
            </p>
          </div>

          <div className="p-4 rounded-2xl border">
            <p className="text-xs opacity-60">
              Sprint Médio
            </p>
            <p className="font-bold">
              {stats.avgSprintDist} m
            </p>
          </div>

          <div className="p-4 rounded-2xl border">
            <p className="text-xs opacity-60">
              Carga Semanal
            </p>
            <p className="font-bold">
              {stats.avgLoad}
            </p>
          </div>

          <div className="p-4 rounded-2xl border">
            <p className="text-xs opacity-60">
              PSE Média
            </p>
            <p className="font-bold">
              {stats.avgPse.toFixed(1)}
            </p>
          </div>

        </div>

        {/* 3 GRÁFICOS NOVOS */}

        <div className="grid grid-cols-1 lg:grid-cols-2 xl:grid-cols-3 gap-6">

          {/* RADAR */}

          <div className="bg-[var(--surface-2)] border rounded-3xl p-6">
            <div className="flex items-center gap-3 mb-6">
              <Brain size={20} />
              <h3 className="font-semibold">
                Perfil Físico Médio
              </h3>
            </div>

            <div className="h-[280px]">
              <ResponsiveContainer>
                <RadarChart data={squadRadarAvg}>
                  <PolarGrid />
                  <PolarAngleAxis dataKey="subject" />

                  <Radar
                    dataKey="A"
                    stroke="#22c55e"
                    fill="#22c55e"
                    fillOpacity={0.3}
                  />

                  <Tooltip content={<CustomTooltip />} />
                </RadarChart>
              </ResponsiveContainer>
            </div>
          </div>

          {/* MATRIZ DE RISCO */}

          <div className="bg-[var(--surface-2)] border rounded-3xl p-6">
            <div className="flex items-center gap-3 mb-6">
              <AlertCircle size={20} />
              <h3 className="font-semibold">
                Matriz de Risco
              </h3>
            </div>

            <div className="h-[280px]">
              <ResponsiveContainer>
                <ScatterChart>
                  <CartesianGrid />

                  <XAxis
                    type="number"
                    dataKey="load"
                  />

                  <YAxis
                    type="number"
                    dataKey="pse"
                  />

                  <ZAxis
                    dataKey="z"
                    range={[100, 100]}
                  />

                  <Tooltip content={<CustomTooltip />} />

                  <Scatter data={riskData}>
                    {riskData.map((entry, index) => (
                      <Cell
                        key={index}
                        fill={
                          entry.pse > 8
                            ? '#ef4444'
                            : '#22c55e'
                        }
                      />
                    ))}
                  </Scatter>
                </ScatterChart>
              </ResponsiveContainer>
            </div>
          </div>

          {/* EVOLUÇÃO DE CARGA */}

          <div className="bg-[var(--surface-2)] border rounded-3xl p-6">
            <div className="flex items-center gap-3 mb-6">
              <Gauge size={20} />
              <h3 className="font-semibold">
                Evolução de Carga
              </h3>
            </div>

            <div className="h-[280px]">
              <ResponsiveContainer>
                <AreaChart data={loadEvolution}>

                  <defs>
                    <linearGradient
                      id="loadGradient"
                      x1="0"
                      y1="0"
                      x2="0"
                      y2="1"
                    >
                      <stop
                        offset="5%"
                        stopColor="#22c55e"
                        stopOpacity={0.4}
                      />

                      <stop
                        offset="95%"
                        stopColor="#22c55e"
                        stopOpacity={0}
                      />
                    </linearGradient>
                  </defs>

                  <CartesianGrid strokeDasharray="3 3" />

                  <XAxis dataKey="jornada" />

                  <YAxis />

                  <Tooltip content={<CustomTooltip />} />

                  <Area
                    type="monotone"
                    dataKey="carga"
                    stroke="#22c55e"
                    fill="url(#loadGradient)"
                    strokeWidth={3}
                  />

                </AreaChart>
              </ResponsiveContainer>
            </div>
          </div>

        </div>

        {/* TOP PERFORMERS */}

        <div>
          <div className="text-xs font-bold mb-4 uppercase">
            Top Performers
          </div>

          <div className="grid grid-cols-1 md:grid-cols-3 gap-4">

            {topPerformers.map((athlete, i) => (

              <Link
                key={athlete.id}
                href={`/atleta/${athlete.id}`}
                className="p-5 border rounded-2xl"
              >

                <div className="flex items-center gap-4">

                  <div className="w-14 h-14 rounded-full bg-zinc-800 flex items-center justify-center font-bold text-white">
                    {athlete.initials}
                  </div>

                  <div className="flex-1">

                    <p className="font-semibold">
                      {athlete.name}
                    </p>

                    <p className="text-xs opacity-60">
                      {athlete.position}
                    </p>

                  </div>

                  <div className="text-right">

                    <p className="font-black text-lg">
                      {athlete.speed.toFixed(1)}
                    </p>

                    <p className="text-xs opacity-50">
                      km/h
                    </p>

                  </div>

                </div>

              </Link>

            ))}

          </div>
        </div>

      </div>
    </div>
  )
}
